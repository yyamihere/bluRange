=== ARTIFACT CREATION QUICK COMMANDS ===

Linux/Mac:
```bash
# Create memory.dump
dd if=/dev/zero bs=1M count=512 of=memory.dump

# Create PCAP with Scapy
python3 << 'EOF'
from scapy.all import *
pkt = IP(dst="192.168.1.100")/TCP(dport=8080)
wrpcap("forensics-lab-01.pcap", [pkt] * 100)
EOF

# Create filesystem.tar.gz
mkdir -p test && tar -czf filesystem.tar.gz test/ && rm -rf test

# Create text files
echo "Apr 15 14:32:01 forensics-lab-01 kernel: Suspicious process detected" > syslog.txt
echo "0 */4 * * * /tmp/fakemalware.sh" > cron_jobs.txt
echo "2026-04-15 14:20:00 http://untrusted-source.lab/system-utility-1.2.3.tar.gz" > browser_history.txt
```
